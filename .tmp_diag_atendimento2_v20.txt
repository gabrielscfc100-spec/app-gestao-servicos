        // Funções Auxiliares de Horário
        function timeToMin(timeStr) {

            if(!timeStr) return 0;

            const [h, m] = timeStr.split(':').map(Number);

            return h * 60 + m;

        }


        function minToTime(min) {

            const h = String(Math.floor(min / 60)).padStart(2, '0');

            const m = String(min % 60).padStart(2, '0');

            return `${
h}
:${
m}
`;

        }


        function obterHojeLocal() {

            const agora = new Date();

            return `${
agora.getFullYear()}
-${
String(agora.getMonth() + 1).padStart(2, '0')}
-${
String(agora.getDate()).padStart(2, '0')}
`;

        }


        function obterMinutosAgoraLocal() {

            const agora = new Date();

            return agora.getHours() * 60 + agora.getMinutes();

        }


        function horarioJaUltrapassado(dataStr, horarioStr) {

            if(!dataStr || !horarioStr) return false;

            const hoje = obterHojeLocal();

            if(dataStr < hoje) return true;

            if(dataStr > hoje) return false;

            return timeToMin(horarioStr) < obterMinutosAgoraLocal();

        }


        function formatarNomeCliente(nome) {

            return (nome || '')
                .trim()
                .replace(/\s+/g, ' ')
                .toLocaleLowerCase('pt-BR')
                .replace(/(^|[\s-])([a-záàâãéêíóôõúç])/gi, (match, separador, letra) => separador + letra.toLocaleUpperCase('pt-BR'));

        }


        function formatarNomeCompletoCliente(nome, sobrenome) {

            return [formatarNomeCliente(nome), formatarNomeCliente(sobrenome)]
                .filter(Boolean)
                .join(' ')
                .trim();

        }


        function validarDataAgendaFiltro() {

            const input = document.getElementById('agenda-data-filtro');

            if (!input) return false;

            if (!input.value) {

                input.value = obterHojeLocal();

            }

            return true;

        }

        function normalizarBuscaCliente(valor) {

            return String(valor || '')
                .normalize('NFD')
                .replace(/[\u0300-\u036f]/g, '')
                .toUpperCase()
                .replace(/\s+/g, ' ')
                .trim();

        }

        function limparSelecaoCliente(contexto) {

        function filtrarClientesBusca(termo) {

            const busca = normalizarBuscaCliente(termo);

            if(!busca) return clientes.slice(0, 12);


            return clientes.filter(c => {

                const codigo = normalizarBuscaCliente(c.codigo || '');

                const nomeCompleto = normalizarBuscaCliente(`${
c.nome || ''}
 ${
c.sobrenome || ''}
`);

                const telefone = normalizarBuscaCliente(c.telefone || '');

                return codigo.includes(busca) || nomeCompleto.includes(busca) || telefone.includes(busca);

            }
).slice(0, 20);

        }

        function localizarClienteUniversal(valor) {

            const busca = (valor || '').trim().toUpperCase();

            if (!busca) return null;


            if (busca.startsWith('#')) {

                const codigoPart = busca.split(' ')[0];

                return clientes.find(c => (c.codigo || '').toUpperCase() === codigoPart);

            }


            const normalizado = busca.replace(/\\s+/g, ' ').trim();

            return clientes.find(c => {

                const nomeCompleto = `${
c.nome || ''}
 ${
c.sobrenome || ''}
`.toUpperCase().replace(/\\s+/g, ' ').trim();

                const codigoNome = `${
c.codigo || ''}
 - ${
nomeCompleto}
`;

                return nomeCompleto === normalizado ||
                    (c.nome || '').toUpperCase() === normalizado ||
                    codigoNome.includes(normalizado) ||
                    normalizado.includes(codigoNome);

            }
) || null;

        }

                : null;

                ? Number(profissionalComissao.comissaoPercentualProdutos == null ? configComissoes.percentualProdutos : profissionalComissao.comissaoPercentualProdutos)
                : null;

            const baseServicosComissao = configComissoes.ativa && configComissoes.sobreServicos
                ? carrinho.filter(i => i.tipoItem !== 'produto').reduce((acc,i)=>acc+Number(i.preco||0),0) : 0;

            const baseProdutosComissao = configComissoes.ativa && configComissoes.sobreProdutos
                ? carrinho.filter(i => i.tipoItem === 'produto').reduce((acc,i)=>acc+Number(i.preco||0),0) : 0;

            const valorComissaoServicos = percentualComissaoServicos == null ? 0 : baseServicosComissao * percentualComissaoServicos / 100;

            const valorComissaoProdutos = percentualComissaoProdutos == null ? 0 : baseProdutosComissao * percentualComissaoProdutos / 100;

            const valorComissaoTotal = valorComissaoServicos + valorComissaoProdutos;


            if(totalVal <= 0) {

                const confirmarZero = confirm('O total deste atendimento está em R$ 0,00. Deseja finalizar mesmo assim?');

                if(!confirmarZero) return;

                        total: totalVal,

                        comissao_percentual_servicos_aplicado: percentualComissaoS

                    }
;

                        empresa_id: empresaAtual.id,
                        atendimento_id: at.id,

                        item_origem_id: (typeof i.id === 'string' && i.id.includes('-')) ? i.id : null,
                        nome: i.nome,
                        quantidade: 1,
                        valor_unitario: Number(i.preco||0),
                        valor_total: Number(i.preco||0)
                    }
));

                    const {
 data: itensSalvos, error: itensErro }
 = await CloudDB.from('itens_atendimento').insert(itensPayload).select();

                    if(itensErro) {

                        await CloudDB.from('atendimentos').delete().eq('id',at.id);

                        throw itensErro;

                    }

                    novaEntrada = entradaCloudParaLocal(at, itensSalvos || []);

                    if(agendamentoOrigemId) novaEntrada.agendamentoId = agendamentoOrigemId;

                }
 else {

                    novaEntrada = {

                        id:Date.now(),
                        clienteId,
                        clienteNome,
                        data:dataVal,
                        pagamento:pagamentoVal,

                        agendamentoId: agendamentoOrigemId || null
                    }
;

                }

                if(agendamentoOrigemId) {

                    concluirAgendamentoAposRecebimento(agendamentoOrigemId, novaEntrada);

                salvarStorage();


                carrinho = [];

                agendamentoEmRecebimentoId = null;

                document.getElementById('busca-cliente').value = '';

                delete document.getElementById('busca-cliente').dataset.clienteId;

                document.getElementById('aviso-recebimento-agenda')?.remove();


                atualizarEspelhoClienteFinanceiro?.();

                atualizarCarrinhoUI();

                atualizarDadosFinanceiros();

                atualizarTabelaClientes();

                renderizarDashboard();

                renderizarAgenda();


                await registrarAuditoria('FINANCEIRO','RECEBIMENTO_REGISTRADO',novaEntrada?.id || null,{
cliente:clienteNome,valor:totalVal,data:dataVal,pagamento:pagamentoVal,agendamento_id:agendamentoOrigemId || null}
);

                alert(agendamentoOrigemId
                    ? 'Atendimento finalizado e receita registrada com sucesso!' : 'Atendimento lançado com sucesso!');

            }
 catch(err) {

                console.error(err);

                alert(`Não foi possível lançar o atendimento: ${
err?.message || 'erro desconhecido'}
`);
