        async function carregarFinanceiroCloud() {
            if(!FINANCEIRO_CLOUD_ATIVO || !CloudDB || !empresaAtual?.cloud || !empresaAtual?.id) return;

            const legados = {
                entradas: entradas.filter(x => !x.cloud),
                saidas: saidas.filter(x => !x.cloud),
                materiais: materiais.filter(x => !x.cloud),
                gastosFixos: gastosFixos.filter(x => !x.cloud)
            };
            Object.entries(legados).forEach(([chave, lista]) => {
                const bk = `barber_${chave}_legacy_backup`;
                if(lista.length && !DataStore.getJSON(bk, null)) DataStore.setJSON(bk, lista);
            });

            const [matRes, fixRes, despRes, atendRes, itensRes] = await Promise.all([
                CloudDB.from('materiais').select('id,empresa_id,nome,ativo,criado_em').eq('empresa_id', empresaAtual.id).order('nome'),
                CloudDB.from('gastos_fixos').select('id,empresa_id,descricao,valor,dia_vencimento,ativo,criado_em,atualizado_em').eq('empresa_id', empresaAtual.id).order('dia_vencimento'),
                CloudDB.from('despesas').select('id,empresa_id,tipo,descricao,material_id,quantidade,valor,data,criado_em,atualizado_em').eq('empresa_id', empresaAtual.id).order('data', { ascending:false }),
                CloudDB.from('atendimentos').select('id,empresa_id,cliente_id,agendamento_id,cliente_nome,data,pagamento,total,profissional_id,comissao_percentual_aplicado,comissao_percentual_servicos_aplicado,comissao_percentual_produtos_aplicado,comissao_base_servicos,comissao_base_produtos,comissao_valor_servicos,comissao_valor_produtos,comissao_valor_total,comissao_status,comissao_repassada_em,criado_em,atualizado_em').eq('empresa_id', empresaAtual.id).order('data', { ascending:false }),
                CloudDB.from('itens_atendimento').select('id,empresa_id,atendimento_id,tipo,item_origem_id,nome,quantidade,valor_unitario,valor_total').eq('empresa_id', empresaAtual.id)

            for(const r of [matRes, fixRes, despRes, atendRes, itensRes]) if(r.error) throw r.error;

        }

        function agendarAtualizacaoFinanceiroCloud() {
            clearTimeout(timerAtualizacaoFinanceiroCloud);
            timerAtualizacaoFinanceiroCloud = setTimeout(async () => {

        }

        function iniciarRealtimeFinanceiroCloud() {
            if(!FINANCEIRO_CLOUD_ATIVO || !CloudDB || !empresaAtual?.cloud || !empresaAtual?.id) return;
            if(canalFinanceiroCloud) { try { CloudDB.removeChannel(canalFinanceiroCloud); } catch(e) {} }
            canalFinanceiroCloud = CloudDB.channel(`financeiro-empresa-${empresaAtual.id}`)
                .on('postgres_changes', {event:'*',schema:'public',table:'atendimentos',filter:`empresa_id=eq.${empresaAtual.id}`}, agendarAtualizacaoFinanceiroCloud)
                .on('postgres_changes', {event:'*',schema:'public',table:'itens_atendimento',filter:`empresa_id=eq.${empresaAtual.id}`}, agendarAtualizacaoFinanceiroCloud)
                .on('postgres_changes', {event:'*',schema:'public',table:'despesas',filter:`empresa_id=eq.${empresaAtual.id}`}, agendarAtualizacaoFinanceiroCloud)
                .on('postgres_changes', {event:'*',schema:'public',table:'materiais',filter:`empresa_id=eq.${empresaAtual.id}`}, agendarAtualizacaoFinanceiroCloud)
                .on('postgres_changes', {event:'*',schema:'public',table:'gastos_fixos',filter:`empresa_id=eq.${empresaAtual.id}`}, agendarAtualizacaoFinanceiroCloud)
                .subscribe();
        }

  function ajustarFinanceiro(){

  function sincronizarBottomNav(){

  }

  else setTimeout(init,0);

        async function carregarFinanceiroCloud() {
            if(!FINANCEIRO_CLOUD_ATIVO || !CloudDB || !empresaAtual?.cloud || !empresaAtual?.id) return;

            const legados = {
                entradas: entradas.filter(x => !x.cloud),
                saidas: saidas.filter(x => !x.cloud),
                materiais: materiais.filter(x => !x.cloud),
                gastosFixos: gastosFixos.filter(x => !x.cloud)
            };
            Object.entries(legados).forEach(([chave, lista]) => {
                const bk = `barber_${chave}_legacy_backup`;
                if(lista.length && !DataStore.getJSON(bk, null)) DataStore.setJSON(bk, lista);
            });

            const [matRes, fixRes, despRes, atendRes, itensRes] = await Promise.all([
                CloudDB.from('materiais').select('id,empresa_id,nome,ativo,criado_em').eq('empresa_id', empresaAtual.id).order('nome'),
                CloudDB.from('gastos_fixos').select('id,empresa_id,descricao,valor,dia_vencimento,ativo,criado_em,atualizado_em').eq('empresa_id', empresaAtual.id).order('dia_vencimento'),
                CloudDB.from('despesas').select('id,empresa_id,tipo,descricao,material_id,quantidade,valor,data,criado_em,atualizado_em').eq('empresa_id', empresaAtual.id).order('data', { ascending:false }),
                CloudDB.from('atendimentos').select('id,empresa_id,cliente_id,agendamento_id,cliente_nome,data,pagamento,total,profissional_id,comissao_percentual_aplicado,comissao_percentual_servicos_aplicado,comissao_percentual_produtos_aplicado,comissao_base_servicos,comissao_base_produtos,comissao_valor_servicos,comissao_valor_produtos,comissao_valor_total,comissao_status,comissao_repassada_em,criado_em,atualizado_em').eq('empresa_id', empresaAtual.id).order('data', { ascending:false }),
                CloudDB.from('itens_atendimento').select('id,empresa_id,atendimento_id,tipo,item_origem_id,nome,quantidade,valor_unitario,valor_total').eq('empresa_id', empresaAtual.id)
            ]);
            for(const r of [matRes, fixRes, despRes, atendRes, itensRes]) if(r.error) throw r.error;

            const itensPorAtendimento = {};
            (itensRes.data || []).forEach(i => (itensPorAtendimento[i.atendimento_id] ||= []).push(i));
            materiais = (matRes.data || []).filter(x => x.ativo !== false).map(materialCloudParaLocal);
            gastosFixos = (fixRes.data || []).filter(x => x.ativo !== false).map(gastoFixoCloudParaLocal);
            saidas = (despRes.data || []).map(saidaCloudParaLocal);
            entradas = (atendRes.data || []).map(a => entradaCloudParaLocal(a, itensPorAtendimento[a.id] || []));

        function gastoFixoCloudParaLocal(row) {
            return {
                id: row.id, empresaId: row.empresa_id, descricao: row.descricao || '',
                valor: Number(row.valor || 0), diaPagamento: Number(row.dia_vencimento || 1),

            };

        function saidaCloudParaLocal(row) {
            const ehMaterial = row.tipo === 'MATERIAL';
            return {
                id: row.id, empresaId: row.empresa_id, descricao: row.descricao || '',
                valor: Number(row.valor || 0), data: row.data || '',
                tipo: ehMaterial ? 'material' : 'outra',
                materialId: row.material_id || null,
                materialNome: row.descricao || '',
                quantidade: row.quantidade == null ? null : Number(row.quantidade),
                cloud: true
            };
        }

        function salvarCacheFinanceiroCloud() {
            DataStore.setJSON('barber_entradas', entradas);
            DataStore.setJSON('barber_saidas', saidas);
            DataStore.setJSON('barber_materiais', materiais);
            DataStore.setJSON('barber_gastos_fixos', gastosFixos);
        }

        async function carregarFinanceiroCloud() {
            if(!FINANCEIRO_CLOUD_ATIVO || !CloudDB || !empresaAtual?.cloud || !empresaAtual?.id) return;

            const legados = {
                entradas: entradas.filter(x => !x.cloud),
                saidas: saidas.filter(x => !x.cloud),
                materiais: materiais.filter(x => !x.cloud),
                gastosFixos: gastosFixos.filter(x => !x.cloud)
            };
            Object.entries(legados).forEach(([chave, lista]) => {
                const bk = `barber_${chave}_legacy_backup`;
                if(lista.length && !DataStore.getJSON(bk, null)) DataStore.setJSON(bk, lista);
            });

            const [matRes, fixRes, despRes, atendRes, itensRes] = await Promise.all([
                CloudDB.from('materiais').select('id,empresa_id,nome,ativo,criado_em').eq('empresa_id', empresaAtual.id).order('nome'),
                CloudDB.from('gastos_fixos').select('id,empresa_id,descricao,valor,dia_vencimento,ativo,criado_em,atualizado_em').eq('empresa_id', empresaAtual.id).order('dia_vencimento'),
                CloudDB.from('despesas').select('id,empresa_id,tipo,descricao,material_id,quantidade,valor,data,criado_em,atualizado_em').eq('empresa_id', empresaAtual.id).order('data', { ascending:false }),
                CloudDB.from('atendimentos').select('id,empresa_id,cliente_id,agendamento_id,cliente_nome,data,pagamento,total,profissional_id,comissao_percentual_aplicado,comissao_percentual_servicos_aplicado,comissao_percentual_produtos_aplicado,comissao_base_servicos,comissao_base_produtos,comissao_valor_servicos,comissao_valor_produtos,comissao_valor_total,comissao_status,comissao_repassada_em,criado_em,atualizado_em').eq('empresa_id', empresaAtual.id).order('data', { ascending:false }),
                CloudDB.from('itens_atendimento').select('id,empresa_id,atendimento_id,tipo,item_origem_id,nome,quantidade,valor_unitario,valor_total').eq('empresa_id', empresaAtual.id)
            ]);
            for(const r of [matRes, fixRes, despRes, atendRes, itensRes]) if(r.error) throw r.error;

            const itensPorAtendimento = {};
            (itensRes.data || []).forEach(i => (itensPorAtendimento[i.atendimento_id] ||= []).push(i));
            materiais = (matRes.data || []).filter(x => x.ativo !== false).map(materialCloudParaLocal);
            gastosFixos = (fixRes.data || []).filter(x => x.ativo !== false).map(gastoFixoCloudParaLocal);
            saidas = (despRes.data || []).map(saidaCloudParaLocal);
            entradas = (atendRes.data || []).map(a => entradaCloudParaLocal(a, itensPorAtendimento[a.id] || []));
            salvarCacheFinanceiroCloud();
            renderizarMateriais();
            renderizarGastosFixos();
            atualizarDadosFinanceiros();
            atualizarTabelaClientes();
            renderizarDashboard();
        }
